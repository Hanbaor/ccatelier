在 TensorFlow 中执行 M - N 时，首先 Python 的减法运算符 - 会被 TensorFlow 重载。接着，TensorFlow 的核心引擎会检查两个矩阵的形状，发现它们不同但兼容，于是自动触发广播机制，在概念上将 M（1x3）和 N（2x1）都扩展成更大的 (2,3) 形状，以便进行运算。但这个过程并不会真的在内存中复制数据以节省资源。然后，这个计算任务被分派给 CPU 或 GPU 进行计算。最后，计算结果被存放在一个新创建的张量中并返回。

```python
import torch
x=torch.tensor(1.0, requires_grad=True)

#y1=x^2 正常计算梯度
y1=x**2
#y2=x^3 但中断梯度追踪
with torch.no_grad():  # 在该上下文管理器中的计算不会追踪梯度
  y2=x**3

y3=y1+y2
y3.backward()
print("梯度dy3/dx:", x.grad)
```

    梯度dy3/dx: tensor(2.)
    

对于公式 y3 = x^2 + x^3，首先通过 y1 = x2 正常记录了第一部分的梯度路径，但使用 tf.stop_gradient(x3) 计算 y2 时，人为地切断了 x 到 y2 在计算图上的梯度连接。因此，从 y3 反向追溯计算梯度时，它只能“看到”y3 通过 y1 对 x 的依赖关系，而 y2 被视为一个与 x 无关的常数，其对 x 的导数为零。最终，总梯度只包含了 x^2 的导数 2x，代入 x=1 后得到的结果便是 2.0，而不是数学上的完整梯度 5.0。

为了方便进行小批量学习，定义数据迭代器


```python
# 第3步：定义数据迭代器
def data_iter(batch_size, features, labels):
    num_examples = len(features)
    indices = list(range(num_examples))
    np.random.shuffle(indices)
    for i in range(0, num_examples, batch_size):
        j = torch.LongTensor(indices[i:min(i + batch_size, num_examples)])
        yield features.index_select(0, j), labels.index_select(0, j)
```

定义模型、损失函数、优化器


```python
# 第4.1步：定义逻辑回归模型
def logistic_model(X, w, b):
    logits = torch.mm(X, w) + b
    return 1 / (1 + torch.exp(-logits))

# 第4.2步：定义二元交叉熵损失函数
def bce_loss(y_hat, y):
    y = y.view(y_hat.size())
    epsilon = 1e-7 # 防止计算 log(0) 出错
    return -torch.mean(y * torch.log(y_hat + epsilon) + (1 - y) * torch.log(1 - y_hat + epsilon))

# 第4.3步：定义随机梯度下降优化器
def sgd(params, lr):
    for param in params:
        param.data -= lr * param.grad
```

定义评估函数来计算当前的准确率和平均损失


```python
# 第5步：定义评估函数
def evaluate_performance(features, labels, w, b, loss_fn):
    with torch.no_grad():
        y_hat = logistic_model(features, w, b)
        loss = loss_fn(y_hat, labels.view(y_hat.size()))
        predicted_classes = (y_hat > 0.5).float()
        accuracy = (predicted_classes == labels.view(predicted_classes.size())).float().mean()
    return accuracy.item(), loss.item()
```

模型参数初始化，给w和b一个随机的初始值

训练模型


```python
# 第7步：训练模型

lr = 0.01
num_epochs = 20
batch_size = 50

train_losses, test_losses = [], []
train_accuracies, test_accuracies = [], []

for epoch in range(num_epochs):
```

```python
#2. 使用 nn.Module 定义模型
from torch import nn

class LogisticRegressionNN(nn.Module):
    def __init__(self, n_features):
        super(LogisticRegressionNN, self).__init__()
        self.linear = nn.Linear(n_features, 1)
        self.sigmoid = nn.Sigmoid()
    def forward(self, x):
        x = self.linear(x)
        x = self.sigmoid(x)
        return x
```

定义评估函数


```python
# 4.定义评估函数
def evaluate_nn(loader, model, criterion):
    model.eval()
    total_loss = 0
    correct = 0
    total = 0
    with torch.no_grad():
        for features, labels in loader:
            outputs = model(features)
            loss = criterion(outputs, labels.view(-1, 1))
            total_loss += loss.item() * len(labels)

            predicted = (outputs > 0.5).float()
            total += len(labels)
            correct += (predicted == labels.view(-1, 1)).sum().item()

    model.train()
    return correct / total, total_loss / total
```


定义模型，损失函数，优化器，初始化模型参数


```python
# 第3.1步：初始化模型参数

num_inputs = 784  # 28 * 28
num_outputs = 10 # 10分类问题

W = torch.tensor(np.random.normal(0, 0.01, (num_inputs, num_outputs)), dtype=torch.float32)
b = torch.zeros(num_outputs, dtype=torch.float32)

W.requires_grad_(requires_grad=True)
b.requires_grad_(requires_grad=True)

# 第3.2步：实现 Softmax 运算
def softmax(X):
    X_exp = torch.exp(X)
    partition = X_exp.sum(dim=1, keepdim=True)
    return X_exp / partition

# 第3.3步：定义模型
def net(X):

    linear_output = torch.mm(X.view((-1, num_inputs)), W) + b
    return softmax(linear_output)

# 第3.4步：定义损失函数 (交叉熵)
def cross_entropy(y_hat, y):
    # y_hat 是模型的输出(batch_size, 10)，y 是真实标签(batch_size)
    # y_hat.gather(1, y.view(-1, 1)) 会挑出每个样本对应正确标签的那个概率值
    return -torch.log(y_hat.gather(1, y.view(-1, 1)))

# 第3.5步：定义优化器
def sgd(params, lr, batch_size):
    for param in params:
        param.data -= lr * param.grad / batch_size
```

定义评估函数


```python
# 第4步：定义精度评估函数
def evaluate_accuracy(data_iter, net_fn):
    acc_sum, n = 0.0, 0
    with torch.no_grad():
        for X, y in data_iter:
            # net_fn(X).argmax(dim=1) 找出预测的类别
            acc_sum += (net_fn(X).argmax(dim=1) == y).float().sum().item()
            n += y.shape[0]
    return acc_sum / n
```

训练模型


```python
# 第5步：训练模型

num_epochs = 10
lr = 0.1

train_losses, test_losses = [], []
train_accuracies, test_accuracies = [], []

def train_epoch(net, train_iter, loss, updater):

    train_l_sum, train_acc_sum, n = 0.0, 0.0, 0
    for X, y in train_iter:
        y_hat = net(X)
        l = loss(y_hat, y)
        l.sum().backward()
        updater([W, b], lr, batch_size)

        W.grad.data.zero_()
        b.grad.data.zero_()

        train_l_sum += l.sum().item()
        train_acc_sum += (y_hat.argmax(dim=1) == y).sum().item()
        n += y.shape[0]
    return train_acc_sum / n, train_l_sum / n

for epoch in range(num_epochs):
    # 训练
    train_acc, train_loss = train_epoch(net, train_iter, cross_entropy, sgd)

    # 在测试集上评估
    test_acc = evaluate_accuracy(test_iter, net)

    train_accuracies.append(train_acc)
    train_losses.append(train_loss)
    test_accuracies.append(test_acc)

    print(f'Epoch {epoch + 1}, Train Loss: {train_loss:.4f}, Train Acc: {train_acc:.3f}, Test Acc: {test_acc:.3f}')

print("训练完成！\n")
```
